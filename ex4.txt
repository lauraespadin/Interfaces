public interface IValidavel
    {
        bool Validar();
    }

 public abstract class CampoFormulario
    {
        public string NomeDoCampo { get; set; }
        public string Valor { get; set; }

        public CampoFormulario(string nomeDoCampo, string valor)
        {
            NomeDoCampo = nomeDoCampo;
            Valor = valor;
        }
    }

 public class CampoTexto : CampoFormulario, IValidavel
    {
        public CampoTexto(string nomeDoCampo, string valor) : base(nomeDoCampo, valor) { }
        public bool Validar()
        {
            if (Valor != "") 
            { 
                return true; 
            }
            else
            {
                return false;
            } 
             
        }

public class CampoNumerico : CampoFormulario, IValidavel
    {
        public CampoNumerico(string nomeDoCampo, string valor) : base(nomeDoCampo, valor) { }
        public bool Validar()
        {
            int numero;
            if (int.TryParse(Valor, out numero))
            {
                return true;
            }
            else
            {
                return false;
            }
        }
    }

 List<IValidavel> campos = new List<IValidavel>();

            CampoTexto campo1 = new CampoTexto("Nome", "Laura");

            CampoTexto campo2 = new CampoTexto("Sobrenome", "Espadin");

            CampoNumerico campo3 = new CampoNumerico("Idade", "16");

            campos.Add(campo1);
            campos.Add(campo2);
            campos.Add(campo3);

            foreach (IValidavel campo in campos)
            {
                if (campo.Validar())
                {
                    Console.WriteLine("Campo válido");
                }
                else
                {
                    Console.WriteLine("Campo inválido");
                }
            }
